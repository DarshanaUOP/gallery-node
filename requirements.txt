# Luminary — Python dependencies
# Install with: pip install -r requirements.txt

# ── Required ─────────────────────────────────────────────────────────────────
Flask>=2.3
flask-cors>=4.0
Pillow>=10.0          # image handling + EXIF extraction
waitress>=2.1         # production WSGI server (replaces Flask's dev server)

# ── Optional — enable extra format support if installed ────────────────────
# HEIC/HEIF (iPhone photos): without this, app.py falls back to pyheif, then

python-dateutil>=2.8

# System tray icon for installed builds (Open Luminary / About / Start-Stop /
# Quit menu — see app/src/backend/tray.py). Not needed for dev-mode runs via
# run.sh, only used when the packaged .exe/binary is launched. Reuses Pillow
# (already required above) for the tray icon image.
pystray>=0.19
#   Linux note: pystray needs a tray backend from the desktop environment
#   itself (AppIndicator/GTK on GNOME, native tray on KDE/XFCE, etc.) — this
#   is a system package, not something pip can install.

# ── Not pip-installable — install separately if you want video metadata ────
# ffmpeg/ffprobe: required for extract_video_metadata(); without it, video
# files are still indexed but without duration/resolution/fps/GPS metadata.
#   macOS:          brew install ffmpeg
#   Ubuntu/Debian:  sudo apt install ffmpeg
#   Windows:        https://ffmpeg.org/download.html

pyinstaller
